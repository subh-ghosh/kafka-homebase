How to Connect

Your permissions allow you to Read, Write, and Create any topic that starts with . You do not need a dashboard to create topics—simply create them directly from your code!

from confluent_kafka import Producer

conf = {
    'bootstrap.servers': 'broker.subartaghosh.co.in:9092',
    'security.protocol': 'SASL_SSL',
    'sasl.mechanisms': 'SCRAM-SHA-512',
    'sasl.username': '',
    'sasl.password': ''
}

producer = Producer(conf)
producer.produce('', key="key", value="hello world")
producer.flush()
const { Kafka } = require('kafkajs')

const kafka = new Kafka({
  clientId: 'my-app',
  brokers: ['broker.subartaghosh.co.in:9092'],
  ssl: true,
  sasl: {
    mechanism: 'scram-sha-512',
    username: '',
    password: ''
  }
})

const producer = kafka.producer()
await producer.connect()
await producer.send({
  topic: '',
  messages: [{ value: 'hello world' }],
})
Properties props = new Properties();
props.put("bootstrap.servers", "broker.subartaghosh.co.in:9092");
props.put("security.protocol", "SASL_SSL");
props.put("sasl.mechanism", "SCRAM-SHA-512");
props.put("sasl.jaas.config", 
  "org.apache.kafka.common.security.scram.ScramLoginModule required " +
  "username=\"\" password=\"\";");

Producer<String, String> producer = new KafkaProducer<>(props);
producer.send(new ProducerRecord<>("", "key", "hello world"));
producer.close();
